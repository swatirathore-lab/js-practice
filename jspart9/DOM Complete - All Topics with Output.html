The DOM tree

#document
  html

    body

TOPIC: Selecting Element by ID

getElementById

let img = document.getElementById("mainImg");
console.log(img);
OUTPUT (console):
<img src="assets/spiderman_img.png" id="mainImg">

Returns exactly ONE element (assuming only one element has that ID on the page).

TOPIC: Selecting Elements by Class Name

getElementsByClassName loop

let smallImages = document.getElementsByClassName("oldImg");
for (let i = 0; i < smallImages.length; i++) {
    smallImages[i].src = "assets/spiderman_img.png";
    console.log(`value of image no. ${i} is changed`);
}
OUTPUT (browser console):
value of image no. 0 is changed
value of image no. 1 is changed
value of image no. 2 is changed
TOPIC: Selecting Elements by Tag Name

getElementsByTagName

let allLinks = document.getElementsByTagName("a");
console.log(allLinks.length);
OUTPUT (console):
5  (if there are 5 <a> tags total on the page)
TOPIC: Query Selectors

querySelector("hi") vs querySelector("h1")

console.dir(document.querySelector("hi"));  // typo - looks for <hi> tag
console.dir(document.querySelector("h1"));  // correct - looks for <h1> tag
OUTPUT (browser console):
querySelector("hi") → null
querySelector("h1") → <h1>Spider Man</h1>
TOPIC: Setting Content in Objects

innerText vs innerHTML

let heading = document.querySelector("h1");
heading.innerText = "New Title";        // plain text only
heading.innerHTML = "<b>New Title</b>";  // parses the <b> tag too
OUTPUT:
innerText result: New Title
innerHTML result: New Title (this one is bold, because the tag got parsed)
TOPIC: Manipulating Attributes

setAttribute / getAttribute

let img = document.querySelector("#mainImg");
img.setAttribute("src", "newImage.png");
console.log(img.getAttribute("src"));
OUTPUT (console):
newImage.png

Image on page also visually updates to the new file.

TOPIC: Manipulating Style (with style attribute)

links[i].style.color = "green"

let links = document.querySelectorAll(".box a");
for (let i = 0; i < links.length; i++) {
    links[i].style.color = "green";
}
OUTPUT (links turn green):
Publisher Stan Lee Steve Ditko

TOPIC: Navigation on Page

parentElement / children / nextElementSibling

let box = document.querySelector(".box");
console.log(box.parentElement);       // body
console.log(box.children);            // [h4, ul] inside box
console.log(box.nextElementSibling);  // whatever comes right after .box
OUTPUT (console):
parentElement → <body>...</body>
children → HTMLCollection(2) [h4, ul]
nextElementSibling → <h2>Creation & Development</h2>
TOPIC: Adding Elements on Page (Practice Qs a, b, c)

Practice Q (a) — Red paragraph

let para1 = document.createElement("p");
para1.innerText = "Hey I'm red";
document.querySelector("body").append(para1);
para1.classList.add("red");
OUTPUT:

Hey I'm red

Practice Q (b) — Blue H3

let H3 = document.createElement("h3");
H3.innerText = "Hey I'm blue H3";
document.querySelector("body").append(H3);
H3.classList.add("blue");
OUTPUT:

Hey I'm blue H3

Practice Q (c) — Nested div (h1 + p inside)

let div = document.createElement("div");
let h1 = document.createElement("h1");
let para2 = document.createElement("p");
h1.innerText = "I'm in a div";
para2.innerText = "ME TOO!";
div.append(h1);
div.append(para2);
div.classList.add("box");
document.querySelector("body").append(div);
OUTPUT:

I'm in a div

ME TOO!
